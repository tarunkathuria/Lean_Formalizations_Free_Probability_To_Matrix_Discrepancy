HigherRankKSRuntime.algorithmic (O : HigherRankKSRuntime.SDPValue.Solver) :
  ∃ C e,
    0 < C ∧
      ∀ (N d r : ℕ) (ε : ℝ) (hr : 1 ≤ r),
        0 ≤ ε →
          ∀ (A : Fin N → Matrix (Fin d) (Fin d) ℂ) (hA : ∀ (i : Fin N), (A i).PosSemidef) (hsum : ∑ i, A i = 1),
            (∀ (i : Fin N), ‖Matrix.toEuclideanCLM (A i)‖ ≤ ε) →
              ∀ (hrank : ∀ (i : Fin N), (A i).rank ≤ r),
                have out := HigherRankKSRuntime.solve O A hr hA hsum hrank;
                (∀ (i : Fin N), out.value i = 1 ∨ out.value i = -1) ∧
                  ‖Matrix.toEuclideanCLM (HigherRankKS.signedSum A out.value)‖ ≤
                      min 1 (10000 * √(ε * Real.log (2 * ↑r))) ∧
                    ↑out.cost ≤ C * (1 + ↑N + ↑d) ^ e
