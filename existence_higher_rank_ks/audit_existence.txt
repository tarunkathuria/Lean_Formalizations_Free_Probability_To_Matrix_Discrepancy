def AugmentedHigherRankKS.SubisotropicExistenceStatement : Prop :=
∀ (N d r : ℕ) (ε : ℝ),
  1 ≤ r →
    0 ≤ ε →
      ∀ (A : Fin N → Matrix (Fin d) (Fin d) ℂ),
        (∀ (i : Fin N), (A i).PosSemidef) →
          ∑ i, A i ≤ 1 →
            (∀ (i : Fin N), ‖Matrix.toEuclideanCLM (A i)‖ ≤ ε) →
              (∀ (i : Fin N), (A i).rank ≤ r) →
                ∃ σ,
                  (∀ (i : Fin N), σ i = 1 ∨ σ i = -1) ∧
                    ‖Matrix.toEuclideanCLM (HigherRankKS.signedSum A σ)‖ ≤ min 1 (10000 * √(ε * Real.log (2 * ↑r)))
